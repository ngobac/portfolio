gfw_automation.py
Selenium tool to automate GeoJSON upload and extract Global Forest Watch share links in bulk
"""
GFW Automation — Part of GEOJSON WORK desktop app
Automates: navigate → close popup → find upload zone →
           upload file → wait zoom → share → extract link
"""
import os, time, threading, requests
import pandas as pd
from pathlib import Path
from datetime import datetime

try:
    from selenium import webdriver
    from selenium.webdriver.chrome.service import Service
    from selenium.webdriver.chrome.options import Options
    from selenium.webdriver.common.by import By
    from selenium.webdriver.support.ui import WebDriverWait
    from selenium.webdriver.support import expected_conditions as EC
    from selenium.common.exceptions import WebDriverException
    SELENIUM_AVAILABLE = True
except ImportError:
    SELENIUM_AVAILABLE = False

TIMING = {
    'PAGE_LOAD': 8, 'POPUP_CLOSE': 5, 'FIND_UPLOAD_ZONE': 5,
    'FILE_UPLOAD_ZOOM': 20, 'FIND_SHARE_BUTTON': 5,
    'SHARE_MODAL_LOAD': 5, 'COPY_LINK': 5, 'MODAL_CLOSE': 3,
}

class GFWAutomatorOptimized:
    def __init__(self):
        self.driver = None
        self.cancelled = False
        self.paused = False
        self.pause_lock = threading.Lock()
        self.stats = {'total':0,'processed':0,'success':0,'failed':0}
        self.excel_file = None
        self.pending_results = []

    def setup_driver(self, chromedriver_path, headless=False):
        service = Service(chromedriver_path)
        options = Options()
        for arg in ['--no-sandbox','--disable-dev-shm-usage','--disable-gpu',
                    '--disable-extensions','--disable-background-networking']:
            options.add_argument(arg)
        if headless:
            options.add_argument('--headless')
        self.driver = webdriver.Chrome(service=service, options=options)
        self.driver.set_page_load_timeout(30)
        self.driver.implicitly_wait(3)
        return {'success': True}

    def step1_navigate(self):
        self.driver.get('https://www.globalforestwatch.org/map/?analysis=eyJzaG93RHJhdyI6dHJ1ZX0%3D')
        time.sleep(TIMING['PAGE_LOAD'])

    def step3_find_upload_zone(self):
        for sel in [(By.ID,'upload-dropzone'),(By.XPATH,"//div[contains(@class,'dropzone')]")]:
            try:
                return WebDriverWait(self.driver, 2).until(EC.presence_of_element_located(sel))
            except: continue
        raise Exception("Upload zone not found")

    def step4_upload_and_wait(self, file_path, upload_element):
        file_input = upload_element.find_element(By.CSS_SELECTOR,'input[type="file"]')
        file_input.send_keys(file_path)
        for _ in range(15):
            if self.cancelled: return False
            time.sleep(1)
        return True

    def step6_get_share_link(self, share_button):
        self.driver.execute_script("arguments[0].click();", share_button)
        WebDriverWait(self.driver, 5).until(
            EC.visibility_of_element_located((By.XPATH,"//div[contains(@class,'c-share')]"))
        )
        time.sleep(TIMING['SHARE_MODAL_LOAD'])
        share_input = WebDriverWait(self.driver, 5).until(
            EC.presence_of_element_located((By.XPATH,"//input[contains(@value,'gfw.global')]"))
        )
        return share_input.get_attribute('value')

    def process_folder(self, folder, excel_file, chromedriver_path):
        if not SELENIUM_AVAILABLE:
            return {'success': False, 'error': 'Selenium not available'}
        self.excel_file = excel_file
        self.setup_driver(chromedriver_path)
        geojson_files = list(Path(folder).glob("*.geojson"))
        for i, f in enumerate(geojson_files):
            if self.cancelled: break
            link = self.upload_and_get_link(str(f))
            status = "SUCCESS" if link else "FAILED"
            self.pending_results.append({
                'GEOJSON Filename': f.name,
                'Share Link': link or 'Failed',
                'Timestamp': datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                'Status': status
            })
        if self.driver: self.driver.quit()
        pd.DataFrame(self.pending_results).to_excel(excel_file, index=False)
        return {'success': True, 'total': len(geojson_files)}

GFWAutomator = GFWAutomatorOptimized

sentinel_hansen_swipe.js — Google Earth Engine
GEE swipe app: compare Sentinel-2 satellite imagery vs Hansen Global Forest Loss (2021–2024). Upload GeoJSON for boundary overlay.
// GEE: SWIPE — SENTINEL-2 vs HANSEN FOREST LOSS
// Author: Ngô Văn Bắc

var app = {};
app.SENTINEL  = 'COPERNICUS/S2_SR_HARMONIZED';
app.HANSEN    = 'UMD/hansen/global_forest_change_2024_v1_12';
app.VIS_S2    = {bands:['B4','B3','B2'], min:0, max:3000, gamma:1.4};
app.CENTER    = {lon:107.191026, lat:10.939236, zoom:10};

app.PERIODS = {
  '2000-2020':{min:0, max:20},  '2021-2024':{min:21, max:24},
  '2021':{min:21,max:21},       '2022':{min:22,max:22},
  '2023':{min:23,max:23},       '2024':{min:24,max:24}
};

app.state = {
  geojson:null, startDate:'2025-01-01',
  endDate:'2025-05-31', maxCloud:30, period:'2021-2024'
};

app.createMaps = function() {
  app.left  = ui.Map(); app.left.setOptions('HYBRID');
  app.right = ui.Map(); app.right.setOptions('SATELLITE');
  ui.Map.Linker([app.left, app.right]);
  app.left.setCenter(app.CENTER.lon, app.CENTER.lat, app.CENTER.zoom);
  app.left.add(ui.Label('◄ Sentinel-2',
    {position:'top-center', backgroundColor:'rgba(0,0,0,0.5)',
     color:'white', fontWeight:'bold', padding:'4px 8px'}));
  app.right.add(ui.Label('Hansen Forest Loss ►',
    {position:'top-center', backgroundColor:'rgba(0,0,0,0.5)',
     color:'white', fontWeight:'bold', padding:'4px 8px'}));
};

app.loadSentinel = function() {
  var col = ee.ImageCollection(app.SENTINEL)
    .filterDate(app.state.startDate, app.state.endDate)
    .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', app.state.maxCloud));
  if (app.state.geojson) col = col.filterBounds(app.state.geojson);
  var composite = col.median();
  app.left.addLayer(composite, app.VIS_S2, 'Sentinel-2');
};

app.loadHansen = function() {
  var p   = app.PERIODS[app.state.period];
  var img = ee.Image(app.HANSEN).select('lossyear');
  var masked = img.updateMask(img.gte(p.min).and(img.lte(p.max)));
  app.right.addLayer(masked,
    {min:p.min, max:p.max, palette:['yellow','orange','red','darkred']},
    'Forest Loss ' + app.state.period);
};

app.loadGeoJSON = function(jsonText) {
  var obj = JSON.parse(jsonText);
  app.state.geojson = ee.FeatureCollection(obj);
  var outline = ee.Image().byte().paint({
    featureCollection: app.state.geojson, color:1, width:1});
  [app.left, app.right].forEach(function(m) {
    m.addLayer(outline, {palette:['00FFFF']}, 'Boundary');
  });
  app.left.centerObject(app.state.geojson, 12);
};

app.boot = function() {
  ui.root.clear();
  app.createMaps();
  ui.root.add(ui.SplitPanel({
    firstPanel:app.left, secondPanel:app.right,
    wipe:true, style:{stretch:'both'}
  }));
  app.loadSentinel();
  app.loadHansen();
};

app.boot();

spatial_forest_db.sql
PostGIS spatial schema for forest plot management + pgvector for AI chatbot knowledge base (Supabase)
-- ── Extensions ─────────────────────────────────────────
CREATE EXTENSION IF NOT EXISTS postgis;
CREATE EXTENSION IF NOT EXISTS vector;

-- ── Forest plots with spatial geometry ─────────────────
CREATE TABLE forest_plots (
  id           SERIAL PRIMARY KEY,
  plot_code    VARCHAR(20) UNIQUE NOT NULL,
  species      VARCHAR(100),
  ndvi_value   NUMERIC(6,4),
  carbon_stock NUMERIC(10,2),   -- tCO2e / ha
  canopy_cover NUMERIC(5,2),    -- percent
  elevation_m  NUMERIC(8,2),
  geom         GEOMETRY(Point, 4326),
  surveyed_at  DATE,
  notes        TEXT
);

-- Spatial index for geometry queries
CREATE INDEX idx_plots_geom ON forest_plots USING GIST (geom);
CREATE INDEX idx_plots_ndvi  ON forest_plots (ndvi_value);

-- ── Deforestation alerts ────────────────────────────────
CREATE TABLE deforestation_alerts (
  id          SERIAL PRIMARY KEY,
  plot_id     INT REFERENCES forest_plots(id),
  alert_date  DATE NOT NULL,
  area_ha     NUMERIC(10,4),
  severity    VARCHAR(20) CHECK (severity IN ('low','medium','high','critical')),
  source      VARCHAR(50),        -- 'Planet' | 'Hansen' | 'Sentinel-2'
  geom        GEOMETRY(Polygon, 4326),
  confirmed   BOOLEAN DEFAULT FALSE
);

CREATE INDEX idx_alerts_geom ON deforestation_alerts USING GIST (geom);

-- ── Vector store for AI chatbot ─────────────────────────
CREATE TABLE knowledge_docs (
  id        BIGSERIAL PRIMARY KEY,
  title     TEXT,
  content   TEXT NOT NULL,
  metadata  JSONB DEFAULT '{}',
  embedding VECTOR(1536),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX ON knowledge_docs
  USING ivfflat (embedding vector_cosine_ops) WITH (lists = 100);

-- Semantic search function
CREATE OR REPLACE FUNCTION match_docs(
  query_embedding VECTOR(1536),
  threshold FLOAT  DEFAULT 0.78,
  match_count INT  DEFAULT 5
)
RETURNS TABLE (id BIGINT, content TEXT, metadata JSONB, similarity FLOAT)
LANGUAGE sql STABLE AS $$
  SELECT id, content, metadata,
         1 - (embedding <=> query_embedding) AS similarity
  FROM   knowledge_docs
  WHERE  1 - (embedding <=> query_embedding) > threshold
  ORDER  BY embedding <=> query_embedding
  LIMIT  match_count;
$$;

-- ── Useful spatial queries ──────────────────────────────

-- High-carbon hotspots (NDVI > 0.6, carbon > 50 tCO2e/ha)
SELECT plot_code, species, ndvi_value, carbon_stock,
       ST_AsGeoJSON(geom) AS geojson
FROM   forest_plots
WHERE  ndvi_value   > 0.6
  AND  carbon_stock > 50
ORDER  BY carbon_stock DESC;

-- Plots within 5 km of an alert polygon
SELECT p.plot_code, p.species,
       ST_Distance(p.geom::geography, a.geom::geography) AS dist_m
FROM   forest_plots p
JOIN   deforestation_alerts a ON ST_DWithin(p.geom::geography, a.geom::geography, 5000)
WHERE  a.severity IN ('high','critical')
ORDER  BY dist_m;

-- Monthly deforestation area summary
SELECT DATE_TRUNC('month', alert_date) AS month,
       source,
       SUM(area_ha) AS total_ha_lost,
       COUNT(*) AS alert_count
FROM   deforestation_alerts
GROUP  BY 1, 2
ORDER  BY 1 DESC;
