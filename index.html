gfw_automation.py

Bulk uploads GeoJSON boundaries to Global Forest Watch and collects the share link for each one. Part of the GEOJSON WORK desktop tool. 

"""GFW Automation. Batch GeoJSON upload and share-link capture."""
import time, threading
import pandas as pd
from pathlib import Path
from datetime import datetime

from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

GFW_MAP = "https://www.globalforestwatch.org/map/"

TIMING = {
    "PAGE_LOAD": 8, "POPUP_CLOSE": 5, "FIND_UPLOAD_ZONE": 5,
    "FILE_UPLOAD_ZOOM": 20, "FIND_SHARE_BUTTON": 5,
    "SHARE_MODAL_LOAD": 5, "COPY_LINK": 5, "MODAL_CLOSE": 3,
}


class GFWAutomator:
    def __init__(self):
        self.driver = None
        self.cancelled = False
        self.pause_lock = threading.Lock()
        self.results = []

    def setup_driver(self, chromedriver_path, headless=False):
        options = Options()
        for flag in ("--no-sandbox", "--disable-dev-shm-usage",
                     "--disable-gpu", "--disable-extensions"):
            options.add_argument(flag)
        if headless:
            options.add_argument("--headless=new")
        self.driver = webdriver.Chrome(
            service=Service(chromedriver_path), options=options)
        self.driver.set_page_load_timeout(30)
        self.driver.implicitly_wait(3)

    def find_upload_zone(self):
        candidates = [
            (By.ID, "upload-dropzone"),
            (By.XPATH, "//div[contains(@class,'dropzone')]"),
        ]
        for locator in candidates:
            try:
                return WebDriverWait(self.driver, 2).until(
                    EC.presence_of_element_located(locator))
            except Exception:
                continue
        raise RuntimeError("upload zone not found")

    def capture_share_link(self):
        button = WebDriverWait(self.driver, 5).until(
            EC.element_to_be_clickable((By.ID, "map-share-button")))
        self.driver.execute_script("arguments[0].click();", button)
        time.sleep(TIMING["SHARE_MODAL_LOAD"])
        field = WebDriverWait(self.driver, 5).until(
            EC.presence_of_element_located(
                (By.XPATH, "//input[contains(@value,'gfw.global')]")))
        return field.get_attribute("value")

    def process_folder(self, folder, excel_path, chromedriver_path):
        self.setup_driver(chromedriver_path)
        files = sorted(Path(folder).glob("*.geojson"))
        for path in files:
            if self.cancelled:
                break
            self.driver.get(GFW_MAP)
            time.sleep(TIMING["PAGE_LOAD"])
            zone = self.find_upload_zone()
            zone.find_element(
                By.CSS_SELECTOR, 'input[type="file"]').send_keys(str(path))
            time.sleep(TIMING["FILE_UPLOAD_ZOOM"])
            try:
                link, status = self.capture_share_link(), "SUCCESS"
            except Exception as exc:
                link, status = f"error: {exc}", "FAILED"
            self.results.append({
                "file": path.name,
                "share_link": link,
                "status": status,
                "captured_at": datetime.now().isoformat(timespec="seconds"),
            })
        self.driver.quit()
        pd.DataFrame(self.results).to_excel(excel_path, index=False)
        return {"processed": len(files)}

sentinel_hansen_swipe.js

Earth Engine application. Splits the view so Sentinel-2 imagery can be swiped against Hansen forest loss for a chosen period. 

// Earth Engine: swipe Sentinel-2 against Hansen forest loss.
// Author: Ngo Van Bac

var app = {};
app.SENTINEL = 'COPERNICUS/S2_SR_HARMONIZED';
app.HANSEN   = 'UMD/hansen/global_forest_change_2024_v1_12';
app.VIS_S2   = {bands: ['B4', 'B3', 'B2'], min: 0, max: 3000, gamma: 1.4};
app.CENTER   = {lon: 107.191026, lat: 10.939236, zoom: 10};

app.PERIODS = {
  '2000-2020': {min: 0,  max: 20},
  '2021-2024': {min: 21, max: 24},
  '2024':      {min: 24, max: 24}
};

app.state = {
  geojson: null,
  startDate: '2025-01-01',
  endDate: '2025-05-31',
  maxCloud: 30,
  period: '2021-2024'
};

app.createMaps = function () {
  app.left = ui.Map();
  app.left.setOptions('HYBRID');
  app.right = ui.Map();
  app.right.setOptions('SATELLITE');
  ui.Map.Linker([app.left, app.right]);
  app.left.setCenter(app.CENTER.lon, app.CENTER.lat, app.CENTER.zoom);
};

app.loadSentinel = function () {
  var collection = ee.ImageCollection(app.SENTINEL)
    .filterDate(app.state.startDate, app.state.endDate)
    .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', app.state.maxCloud));

  if (app.state.geojson) {
    collection = collection.filterBounds(app.state.geojson);
  }
  app.left.addLayer(collection.median(), app.VIS_S2, 'Sentinel-2');
};

app.loadHansen = function () {
  var period = app.PERIODS[app.state.period];
  var lossYear = ee.Image(app.HANSEN).select('lossyear');
  var masked = lossYear.updateMask(
    lossYear.gte(period.min).and(lossYear.lte(period.max)));

  app.right.addLayer(masked, {
    min: period.min,
    max: period.max,
    palette: ['yellow', 'orange', 'red', 'darkred']
  }, 'Forest loss ' + app.state.period);
};

app.loadBoundary = function (geojsonText) {
  app.state.geojson = ee.FeatureCollection(JSON.parse(geojsonText));
  var outline = ee.Image().byte().paint({
    featureCollection: app.state.geojson, color: 1, width: 1
  });
  [app.left, app.right].forEach(function (map) {
    map.addLayer(outline, {palette: ['00FFFF']}, 'Boundary');
  });
  app.left.centerObject(app.state.geojson, 12);
};

app.boot = function () {
  ui.root.clear();
  app.createMaps();
  ui.root.add(ui.SplitPanel({
    firstPanel: app.left,
    secondPanel: app.right,
    wipe: true,
    style: {stretch: 'both'}
  }));
  app.loadSentinel();
  app.loadHansen();
};

app.boot();

forest_spatial.sql

PostGIS schema behind the plot inventory, plus the pgvector store that backs a Vietnamese language assistant for field staff. 

CREATE EXTENSION IF NOT EXISTS postgis;
CREATE EXTENSION IF NOT EXISTS vector;

-- Inventory plots -------------------------------------------------
CREATE TABLE forest_plots (
    id            serial PRIMARY KEY,
    plot_code     varchar(20) UNIQUE NOT NULL,
    species       varchar(100),
    ndvi_value    numeric(6,4),
    carbon_stock  numeric(10,2),          -- tCO2e per hectare
    canopy_cover  numeric(5,2),           -- percent
    geom          geometry(Point, 4326),
    surveyed_at   date NOT NULL
);

CREATE INDEX forest_plots_geom_idx ON forest_plots USING gist (geom);
CREATE INDEX forest_plots_ndvi_idx ON forest_plots (ndvi_value);

-- Change detection alerts ----------------------------------------
CREATE TABLE loss_alerts (
    id          serial PRIMARY KEY,
    alert_date  date NOT NULL,
    area_ha     numeric(10,4),
    severity    varchar(10) CHECK (severity IN ('low','medium','high')),
    source      varchar(30),              -- Planet | Sentinel-2 | Hansen
    geom        geometry(Polygon, 4326),
    confirmed   boolean DEFAULT false
);

CREATE INDEX loss_alerts_geom_idx ON loss_alerts USING gist (geom);

-- Which plots sit inside a confirmed loss polygon? ----------------
SELECT p.plot_code,
       p.species,
       p.carbon_stock,
       a.alert_date,
       a.severity,
       round(a.area_ha, 2) AS area_ha
FROM   forest_plots p
JOIN   loss_alerts  a ON ST_Within(p.geom, a.geom)
WHERE  a.confirmed
  AND  a.alert_date >= date '2024-01-01'
ORDER  BY p.carbon_stock DESC;

-- Plots within 5 km of any high severity alert --------------------
SELECT DISTINCT p.plot_code,
       round(min(ST_Distance(p.geom::geography,
                             a.geom::geography))::numeric, 0) AS metres
FROM   forest_plots p
JOIN   loss_alerts  a
  ON   ST_DWithin(p.geom::geography, a.geom::geography, 5000)
WHERE  a.severity = 'high'
GROUP  BY p.plot_code
ORDER  BY metres;

-- Knowledge base for the field assistant --------------------------
CREATE TABLE knowledge_docs (
    id         bigserial PRIMARY KEY,
    title      text,
    content    text NOT NULL,
    metadata   jsonb DEFAULT '{}',
    embedding  vector(1536),
    created_at timestamptz DEFAULT now()
);

CREATE INDEX ON knowledge_docs
    USING ivfflat (embedding vector_cosine_ops) WITH (lists = 100);

CREATE OR REPLACE FUNCTION match_docs(
    query_embedding vector(1536),
    threshold       float DEFAULT 0.78,
    match_count     int   DEFAULT 5
)
RETURNS TABLE (id bigint, title text, content text, similarity float)
LANGUAGE sql STABLE AS $$
    SELECT id,
           title,
           content,
           1 - (embedding <=> query_embedding) AS similarity
    FROM   knowledge_docs
    WHERE  1 - (embedding <=> query_embedding) > threshold
    ORDER  BY embedding <=> query_embedding
    LIMIT  match_count;
$$;
